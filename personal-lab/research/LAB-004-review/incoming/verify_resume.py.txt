def req(port: int, payload: dict, timeout: float = 45.0) -> tuple[int, dict]:
    data = json.dumps(payload, ensure_ascii=False).encode('utf-8')
    request = urllib.request.Request(
        f'http://127.0.0.1:{port}/',
        data=data,
        headers={'Content-Type': 'application/json'},
    )
    opener = urllib.request.build_opener(urllib.request.ProxyHandler({}))
    try:
        with opener.open(request, timeout=timeout) as resp:
            return resp.status, json.loads(resp.read().decode('utf-8'))
    except urllib.error.HTTPError as err:
        body = err.read().decode('utf-8')
        try:
            return err.code, json.loads(body)
        except Exception:
            return err.code, {'raw_error': body}


def find_element(obs: dict, *, label_sub: str | None = None, exact_label: str | None = None, el_type: str | None = None) -> str:
    for item in obs.get('elements', []):
        if el_type and item.get('type') not in ((el_type, 'input') if el_type == 'text' else (el_type,)):
            continue
        lbl = item.get('label', '')
        if exact_label is not None and lbl == exact_label:
            return item['element']
        if label_sub is not None and label_sub in lbl:
            return item['element']
        if exact_label is None and label_sub is None and el_type is not None:
            return item['element']
    raise KeyError(f'Element not found (label_sub={label_sub}, exact_label={exact_label}, el_type={el_type}) in {[e.get("label") for e in obs.get("elements", [])]}')


def start_daemon(
    run_dir: Path,
    port: int,
    world: Path = WORLD_PILOT,
    resume: bool = False,
    read_only: bool = False,
    app_dir: Path | None = None,
    profiles_dir: Path | None = None,
    fail_after_ui_commit_step: int | None = None,
) -> subprocess.Popen:
    cmd = [
        sys.executable,
        str(DRIVER_PY),
        'serve',
        '--world',
        str(world),
        '--run-dir',
        str(run_dir),
        '--port',
        str(port),
    ]
    if resume:
        cmd.append('--resume')
    if read_only:
        cmd.append('--read-only')
    if app_dir is not None:
        cmd.extend(['--app-dir', str(app_dir)])
    if profiles_dir is not None:
        cmd.extend(['--profiles-dir', str(profiles_dir)])
    if fail_after_ui_commit_step is not None:
        cmd.extend(['--fail-after-ui-commit-step', str(fail_after_ui_commit_step)])

    proc = subprocess.Popen(
        cmd,
        cwd=str(REPO_ROOT),
        stdout=subprocess.PIPE,
        stderr=subprocess.PIPE,
        text=True,
    )
    deadline = time.time() + 15.0
    while time.time() < deadline:
        if proc.poll() is not None:
            out, err = proc.communicate(timeout=2)
            raise RuntimeError(f'Daemon exited early with code {proc.returncode}: stdout={out} stderr={err}')
        line = proc.stdout.readline()
        if line:
            info = json.loads(line)
            if info.get('status') == 'listening':
                return proc
        time.sleep(0.05)
    proc.kill()
    raise TimeoutError('Timed out waiting for daemon to listen')


def stop_daemon(proc: subprocess.Popen) -> dict:
    if proc.poll() is None:
        proc.send_signal(signal.SIGTERM)
        try:
            out, err = proc.communicate(timeout=15)
        except subprocess.TimeoutExpired:
            proc.kill()
            out, err = proc.communicate(timeout=5)
    else:
        out, err = proc.communicate(timeout=5)
    return {'returncode': proc.returncode, 'stdout': out, 'stderr': err}


def snapshot_core_hashes(run_dir: Path) -> dict[str, str]:
    result = {}
    for name in ('run.json', 'world.json', 'summary.json', 'checkpoint.json', 'events.jsonl', 'source-manifest.json'):
        p = run_dir / name
        if p.exists():
            result[name] = hashlib.sha256(p.read_bytes()).hexdigest()
    return result


def verify_playwright_api_contract() -> dict:
    """Verify that Driver invokes the exact Playwright 1.62.0 sync_api method signatures when HAS_PLAYWRIGHT is True."""
    import importlib.util
    calls = []

    class MockElement:
        def as_element(self):
            return self
        def click(self):
            calls.append('element.click')
        def fill(self, v):
            calls.append(f'element.fill:{v}')
        def select_option(self, label=None):
            calls.append(f'element.select_option:{label}')
        def set_checked(self, v):
            calls.append(f'element.set_checked:{v}')

    class MockHandle:
        def get_property(self, idx):
            return MockElement()
        def dispose(self):
            calls.append('handle.dispose')

    class MockClock:
        def install(self, time=None):
            calls.append(f'page.clock.install:{time.isoformat() if hasattr(time, "isoformat") else time}')

    class MockMouse:
        def move(self, x, y):
            calls.append(f'page.mouse.move:{x},{y}')
        def wheel(self, dx, dy):
            calls.append(f'page.mouse.wheel:{dx},{dy}')

    class MockPage:
        viewport_size = {'width': 390, 'height': 844}
        clock = MockClock()
        mouse = MockMouse()
        def set_default_timeout(self, ms):
            calls.append(f'page.set_default_timeout:{ms}')
        def on(self, ev, cb):
            calls.append(f'page.on:{ev}')
        def goto(self, url, wait_until=None):
            calls.append(f'page.goto:{wait_until}')
        def wait_for_timeout(self, ms):
            pass
        def evaluate(self, script):
            if 'localStorage' in script:
                return {'lifegame-personal-lab-my-v1': json.dumps({'lastProcessed': '2026-11-16', 'revision': 1, 'ledger': []})}
            return {
                'title': 'LifeGame Lab',
                'url': 'http://127.0.0.1:8041/personal-lab/',
                'virtual_time_observed': '2026-11-16T18:00:00.000Z',
                'modal': False,
                'viewport': {'width': 390, 'height': 844},
                'visible_text': 'LifeGame Lab',
                'elements': [{'element': 'e1', 'label': 'Начать', 'type': 'button', 'disabled': False, 'rect': {'x': 10, 'y': 10, 'width': 50, 'height': 20}}],
            }
        def evaluate_handle(self, expr):
            calls.append(f'page.evaluate_handle:{expr}')
            return MockHandle()
        def screenshot(self, path=None, animations=None):
            calls.append(f'page.screenshot:{animations}')
            Path(path).write_bytes(b'\x89PNG\r\n\x1a\n')
        def close(self):
            calls.append('page.close')

    class MockTracing:
        def start(self, screenshots=True, snapshots=True, sources=False):
            calls.append(f'tracing.start:{screenshots},{snapshots},{sources}')
        def start_chunk(self, title=''):
            calls.append(f'tracing.start_chunk:{title}')
        def stop_chunk(self, path=None):
            calls.append('tracing.stop_chunk')
            with zipfile.ZipFile(path, 'w') as zf:
                zf.writestr('trace.trace', '{}\n')
        def stop(self):
            calls.append('tracing.stop')

    class MockContext:
        tracing = MockTracing()
        pages = []
        def route(self, pattern, handler):
            calls.append(f'context.route:{pattern}')
        def set_offline(self, offline):
            calls.append(f'context.set_offline:{offline}')
        def new_page(self):
            calls.append('context.new_page')
            return MockPage()
        def close(self):
            calls.append('context.close')

    class MockPW:
        class chromium:
            @staticmethod
            def launch_persistent_context(user_data_dir, **kwargs):
                calls.append('chromium.launch_persistent_context')
                return MockContext()
        def start(self):
            calls.append('pw.start')
            return self
        def stop(self):
            calls.append('pw.stop')

    spec = importlib.util.spec_from_file_location('drv_contract', DRIVER_PY)
    mod = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(mod)
    mod.HAS_PLAYWRIGHT = True
    mod._pw_sync_playwright = lambda: MockPW()

    with tempfile.TemporaryDirectory(prefix='lg_pw_contract_') as tmp:
        d = mod.Driver(WORLD_PILOT, Path(tmp) / 'run')
        res = d.session('P01', 1)
        obs_id = res['observation']['observation_id']
        d.action({'op': 'act', 'actor': 'P01', 'observation_id': obs_id, 'element': 'e1', 'action': 'click', 'intent': 'test'})
        d.end('P01', outcome='done', memory='mem')
        d.finish()

    required = [
        'pw.start',
        'chromium.launch_persistent_context',
        'context.route:**/*',
        'tracing.start:True,True,False',
        'context.set_offline:False',
        'tracing.start_chunk:P01 day 1',
        'context.new_page',
        'page.clock.install:2026-11-16T21:00:00+03:00',
        'page.goto:networkidle',
        'element.click',
        'tracing.stop_chunk',
        'page.close',
        'tracing.stop',
        'context.close',
        'pw.stop',
    ]
    missing = [r for r in required if r not in calls]
    return {'pass': not missing, 'verified_calls': required, 'missing': missing}

        # 5g. Opening finished run with --resume --read-only allows audit/brief and blocks session/mutations
        daemon = start_daemon(run_dir, port, world=WORLD_PILOT, resume=True, read_only=True)
        code_ro_audit, ro_audit = req(port, {'op': 'audit'})
        assert code_ro_audit == 200 and ro_audit['finished'] is True
        code_ro_mut, ro_mut = req(port, {'op': 'session', 'actor': 'P01', 'day': 7})
        assert code_ro_mut == 400 and 'read-only' in ro_mut.get('transport_error', ''), ro_mut
        stop_daemon(daemon)
        daemon = None
        assert snapshot_core_hashes(run_dir) == hashes_after_finish, 'Evidence modified during read-only inspection!'

        report['criteria']['5_guards_world_version_lock_finished'] = {
            'pass': True,
            'without_resume_error': no_resume_proc.stderr.strip().splitlines()[-1],
            'world_mismatch_error': world_mismatch_proc.stderr.strip().splitlines()[-1],
            'app_version_mismatch_error': app_mismatch_proc.stderr.strip().splitlines()[-1],
            'busy_run_dir_error': busy_run_proc.stderr.strip().splitlines()[-1],
            'busy_profile_error': busy_prof_proc.stderr.strip().splitlines()[-1],
            'finished_run_resume_error': finished_resume_proc.stderr.strip().splitlines()[-1],
            'read_only_mutation_error': ro_mut.get('transport_error'),
            'evidence_hashes_unchanged_across_all_rejections': True,
        }

        # =====================================================================
        # PHASE 7: Budget exhaustion check & analyze_run.py schema + trace validation
        # =====================================================================
        # Quick budget exhaustion check on a 2-step budget world in budget_dir
        small_world = json.loads(WORLD_PILOT.read_text())
        small_world['sessions']['P01']['1']['budget'] = 2
        small_world_path = verif_root / 'world.budget-check.json'
        small_world_path.write_text(json.dumps(small_world, ensure_ascii=False, indent=2) + '\n')

        daemon = start_daemon(budget_dir, 8057, world=small_world_path, resume=False)
        code, r_b = req(8057, {'op': 'session', 'actor': 'P01', 'day': 1})
        obs_b = r_b['observation']
        el = find_element(obs_b, label_sub='Начать свою жизнь в LifeGame')
        code, r_b1 = req(8057, {'op': 'act', 'actor': 'P01', 'observation_id': obs_b['observation_id'], 'element': el, 'action': 'click', 'intent': 'step 1 of 2'})
        obs_b = r_b1['observation']

        # Stop & resume after 1 of 2 budget steps used
        stop_daemon(daemon)
        daemon = start_daemon(budget_dir, 8057, world=small_world_path, resume=True)
        code, obs_b = req(8057, {'op': 'observe', 'actor': 'P01'})
        assert obs_b['actions_used'] == 1 and obs_b['action_budget'] == 2
        el = find_element(obs_b, exact_label='Привычки')
        code, r_b2 = req(8057, {'op': 'act', 'actor': 'P01', 'observation_id': obs_b['observation_id'], 'element': el, 'action': 'click', 'intent': 'step 2 of 2'})
        obs_b = r_b2['observation']
        assert obs_b['actions_used'] == 2

        # Step 3 must return action_budget_exhausted
        el = find_element(obs_b, label_sub='+ Новое')
        code, r_b3 = req(8057, {'op': 'act', 'actor': 'P01', 'observation_id': obs_b['observation_id'], 'element': el, 'action': 'click', 'intent': 'step 3 over budget'})
        assert code == 200 and r_b3.get('error') == 'action_budget_exhausted' and r_b3.get('actions_used') == 2, r_b3
        code, _ = req(8057, {'op': 'end', 'actor': 'P01', 'outcome': 'Budget exhausted after 2 steps', 'memory': 'Budget test ok'})
        assert code == 200
        req(8057, {'op': 'finish'})
        stop_daemon(daemon)
        daemon = None
        small_world_path.unlink(missing_ok=True)

        # Run analyze_run.py on resume-test-001
        analyze_proc = subprocess.run(
            [sys.executable, str(ANALYZE_PY), str(run_dir)],
            cwd=str(REPO_ROOT),
            capture_output=True,
            text=True,
            check=True,
        )
        facts_report = json.loads((run_dir / 'facts-report.json').read_text())
        assert facts_report['all_captured_personal_states_schema_valid'] is True, facts_report
        assert facts_report['all_captured_namespaces_lab_only'] is True, facts_report
        assert facts_report['profile_directories_distinct'] is True, facts_report
        assert facts_report['runtime_files_changed_during_run'] == [], facts_report

        # Also validate all trace-resume-*.zip continuation archives
        continuation_traces = []
        for tpath in sorted(run_dir.glob('P*/day-*/trace-resume-*.zip')):
            with zipfile.ZipFile(tpath) as zf:
                continuation_traces.append(
                    {
                        'path': str(tpath),
                        'bytes': tpath.stat().st_size,
                        'zip_valid': zf.testzip() is None,
                        'has_trace': any(n.endswith('.trace') for n in zf.namelist()),
                    }
                )
        assert continuation_traces and all(t['zip_valid'] and t['has_trace'] for t in continuation_traces), continuation_traces

        # Verify Playwright 1.62.0 API contract
        pw_contract = verify_playwright_api_contract()
        assert pw_contract['pass'] is True, pw_contract

        # Verify original pilot evidence remains 100% intact
        orig_verify = subprocess.run(
            [sys.executable, str(VERIFY_EVIDENCE_PY)],
            cwd=str(REPO_ROOT),
            capture_output=True,
            text=True,
            check=True,
        )

        report['criteria']['6_traces_snapshots_schema_and_error_control'] = {
            'pass': True,
            'all_captured_personal_states_schema_valid': facts_report['all_captured_personal_states_schema_valid'],
            'all_captured_namespaces_lab_only': facts_report['all_captured_namespaces_lab_only'],
            'runtime_files_changed_during_run': facts_report['runtime_files_changed_during_run'],
            'primary_traces_validated': len(facts_report['traces']),
            'continuation_traces_validated': continuation_traces,
            'resume_event_segments': sorted(str(p) for p in run_dir.glob('events.resume-*.jsonl')),
            'stale_observation_rejected': r_stale.get('transport_error'),
            'unknown_element_error_handled': r_bad_el['error'],
            'budget_exhaustion_handled': r_b3,
            'playwright_api_contract': pw_contract,
            'original_evidence_verify_stdout': orig_verify.stdout.strip(),
        }

        report['all_criteria_passed'] = all(c.get('pass') for c in report['criteria'].values())
        report_path = verif_root / 'resume-test-001' / 'verification-report.json'
        report_path.write_text(json.dumps(report, ensure_ascii=False, indent=2) + '\n')
        print(json.dumps({'all_criteria_passed': report['all_criteria_passed'], 'report': str(report_path)}, ensure_ascii=False))
    finally:
        if daemon is not None:
            stop_daemon(daemon)
        http_srv.kill()
        http_srv.wait()


if __name__ == '__main__':
    main()
